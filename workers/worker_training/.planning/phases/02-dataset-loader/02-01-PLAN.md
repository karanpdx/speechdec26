<!-- .npz file schema produced by stubs/data_stubs.py -->

EEG / MEG file (sub-XX_epochs.npz):
  data:            float32 (n_epochs, n_channels, n_timepoints)
  labels:          str array (n_epochs,)
  subject_id:      str scalar
  sfreq:           float64 scalar
  ch_names:        str array (n_channels,)
  event_onsets_s:  float64 array (n_epochs,)

fMRI file (sub-XX_betas.npz):
  data:            float32 (n_words, n_voxels)
  labels:          str array (n_words,)
  subject_id:      str scalar
  voxel_coords:    int32 (n_voxels, 3)

vocab_embeddings.npz:
  vocab:           str array (V,)
  embeddings:      float32 (V, 768)

split_v1.json:
  {
    "train": {"eeg": ["sub-01_epochs.npz", ...], "meg": [...], "fmri": [...]},
    "val":   { ... },
    "test":  { ... }
  }

MultiModalDataset item schema (what __getitem__ must return):
  modality:    str         — 'eeg', 'meg', or 'fmri'
  data:        FloatTensor — (n_channels, n_timepoints) for EEG/MEG; (n_voxels,) for fMRI
  label:       str         — word string
  label_idx:   int         — index into vocabulary
  bert_emb:    FloatTensor — (768,)
  subject_id:  str
  subject_idx: int         — index for SubjectEmbedding lookup
