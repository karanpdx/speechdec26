    Replace the `collate_fn` body (currently `raise NotImplementedError`).

    Implementation outline:
      groups: dict[str, list[dict]] = {}
      for item in batch:
          groups.setdefault(item["modality"], []).append(item)

      output = {}
      for modality, items in groups.items():
          output[modality] = {
              "data": torch.stack([i["data"] for i in items]),
              "label_idx": torch.tensor([i["label_idx"] for i in items], dtype=torch.long),
              "bert_emb": torch.stack([i["bert_emb"] for i in items]),
              "subject_idx": torch.tensor([i["subject_idx"] for i in items], dtype=torch.long),
              "labels": [i["label"] for i in items],
          }

      output["shared_label_mask"] = build_shared_label_mask(batch)
      return output
